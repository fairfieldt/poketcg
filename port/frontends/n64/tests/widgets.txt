# The request UIs that random play rarely reaches, fed as synthetic requests
# (TCGN64_SELFTEST, host builds only) before the player's first duel menu of a
# real duel against the bot, then answered with scripted presses.
#env TCGN64_SELFTEST=1
wait 5
press RIGHT
press START
until ASK~yes_no~PLAYER 100000 3 P1
shot yes_no
press DOWN
press A
expect REQ~yes_no~PLAYER~n=2~->~0
shot two_items_deck_choice
press RIGHT
press A
expect REQ~menu/two_items~PLAYER~n=2~->~1
shot color
repeat 2 press DOWN
press A
expect REQ~color~PLAYER~n=6~->~3
shot number
repeat 2 press UP
press A
expect REQ~number~PLAYER~n=10~->~5
shot generic_menu
press DOWN
press A
expect REQ~menu/generic~PLAYER~n=3~->~1
shot two_items_menu
press B
press A
expect REQ~menu/two_items~PLAYER~n=2~->~0
expect_none REQ~menu/two_items~PLAYER~n=2~->~-1
shot card_list_menu
press DOWN
press A
expect REQ~menu/card_list~PLAYER
expect_none REQ~menu/card_list~PLAYER~n=0
shot prize
press LEFT
press A
press RIGHT
press A
expect REQ~prize~PLAYER~n=6~->~1
shot confirm
press A
press B
expect REQ~confirm~PLAYER~n=0~->~0
expect REQ~confirm~PLAYER~n=0~->~-1
# Peek: the other play area, then the hand (prize 0 is taken: A does nothing there)
shot peek_area
press DOWN
press A
expect REQ~menu/peek_area~PLAYER~n=2~->~1
shot peek_target
press A
press LEFT
press A
expect REQ~menu/peek_target~PLAYER~n=8~->~6
press B
expect REQ~menu/peek_target~PLAYER~n=8~->~-1
# as if the other side were the engine AI: its confirm and view-only list go to controller 1
shot confirm_for_ai
press A
expect ASK~confirm(ai)~BOT
expect REQ~confirm~BOT~n=0~->~0
shot view_only_list_for_ai
press DOWN
press B
expect ASK~card_list(ai)~BOT
# the random bot leaves its own view-only list at once
expect REQ~card_list~BOT~n=7~->~-1 2
expect ASK~duel_menu~PLAYER
monkey 2000 4 P1
